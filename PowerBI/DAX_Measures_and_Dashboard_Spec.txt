MEASURES
--------
Total Employees = COUNTROWS('HR_Employee_Attrition')

Employees Left = CALCULATE([Total Employees], 'HR_Employee_Attrition'[Attrition] = "Yes")

Employees Stayed = CALCULATE([Total Employees], 'HR_Employee_Attrition'[Attrition] = "No")

Attrition Rate = DIVIDE([Employees Left], [Total Employees], 0)

Average Monthly Income = AVERAGE('HR_Employee_Attrition'[MonthlyIncome])
